    def _worker(self, payload):
        try:
            self._post_records(AIRTABLE_SNAPSHOTS_TABLE, [self._snapshot_record(payload)])
            sigs = self._signal_records(payload)
            if sigs:
                self._post_records(AIRTABLE_SIGNALS_TABLE, sigs)
            last_trade_id = int(store.get("airtable_last_trade_id", 0) or 0)
            while True:
                rows = self._trade_rows_after(last_trade_id, 100)
                if not rows:
                    break
                records = [self._trade_record(r) for r in rows]
                self._post_records(AIRTABLE_TRADES_TABLE, records)
                last_trade_id = int(rows[-1][0])
                store.set("airtable_last_trade_id", last_trade_id)
                if len(rows) < 100:
                    break
            with self.lock:
                self.last_ok_ts = time.time()
                self.last_error = ""
                self.sent_snapshots += 1
            store.event("AIRTABLE_OK", f"snapshot synced; trades through {last_trade_id}")
        except Exception as e:
            with self.lock:
                self.last_error = str(e)[:500]
            store.event("AIRTABLE_ERROR", self.last_error)
        finally:
            with self.lock:
                self.syncing = False

    def maybe_sync(self, day_results, swing_results, books, regime, breadth):
        if not self.token():
            return
        now = time.time()
        with self.lock:
            if self.syncing:
                return
            if self.last_snapshot_ts and now - self.last_snapshot_ts < TELEMETRY_SNAPSHOT_SECONDS:
                return
            self.last_snapshot_ts = now
            self.syncing = True
        payload = {
            "ts": now,
            "day_results": [dict(a) for a in day_results[:12]],
            "swing_results": [dict(a) for a in swing_results[:8]],
            "books": {k: dict(v) for k, v in books.items()},
            "regime": regime,
            "breadth": breadth,
        }
        threading.Thread(target=self._worker, args=(payload,), daemon=True).start()

def reset_swing_confirmation(market, reason="reset"):
    st = swing_confirmation_state[market]
    st.update({"count":0,"first_ts":0.0,"last_ts":0.0,"last_score":None,"last_timing":None,"scores":[],"timings":[],"last_reason":reason})

def update_swing_confirmation(a, eligible, now=None):
    now = time.time() if now is None else now
    market = a["market"]
    st = swing_confirmation_state[market]
    if not eligible:
        reset_swing_confirmation(market, entry_reject_reason(a, "SWING")); return 0
    if st.get("count",0) and now - st.get("last_ts",0.0) > SWING_CONFIRMATION_TIMEOUT_SECONDS:
        reset_swing_confirmation(market,"timeout"); st = swing_confirmation_state[market]
    if st.get("last_score") is not None and a.get("score",0) < float(st["last_score"]) - SWING_CONFIRMATION_SCORE_DROP_RESET:
        reset_swing_confirmation(market,"score drop"); st = swing_confirmation_state[market]
    if st.get("last_timing") is not None and a.get("timing_score",0) < float(st["last_timing"]) - SWING_CONFIRMATION_TIMING_DROP_RESET:
        reset_swing_confirmation(market,"timing drop"); st = swing_confirmation_state[market]
    if st.get("count",0) == 0 or now - st.get("last_ts",0.0) >= SWING_CONFIRMATION_MIN_SECONDS:
        if st.get("count",0) == 0: st["first_ts"] = now
